จับคู่ลูกหนี้ UC OPD กับ STM

อัปโหลดไฟล์ลูกหนี้รายตัวจาก HOSxP และไฟล์ STM OPUCS จาก สปสช. ระบบจับคู่ด้วย เลขบัตรประชาชน + วันที่รับบริการ แล้วแสดงลูกหนี้ที่ยังไม่พบใน statement ให้ดาวน์โหลด ไฟล์ทั้งหมดประมวลผลในเครื่องของคุณ ไม่ถูกส่งออกไปที่ใด

คู่มือ: ดึงข้อมูลลูกหนี้จาก HOSxP (PostgreSQL) และจับคู่กับ STM
  1. เปิดเครื่องมือรัน SQL ที่ต่อฐาน HOSxP XE (PostgreSQL) เช่น หน้า SQL Query ใน HOSxP, DBeaver, pgAdmin หรือ Navicat ใช้ user ที่อ่านตารางได้อย่างน้อย ovst, patient, vn_stat, visit_pttype, pttype, ovstdiag, opitemrece, nondrugitems, drugitems
  2. คัดลอก SQL ด้านล่าง แล้วแก้ช่วงวันที่ใน CTE params บรรทัดบนสุด: d1 = วันเริ่ม, d2 = วันสิ้นสุด (รูปแบบ ค.ศ. YYYY-MM-DD) แนะนำให้ดึงทีละเดือนหรือทีละไตรมาส
  3. รัน SQL ได้ 1 แถวต่อ visit เฉพาะ visit OPD คนไทยที่ไม่ admit, มีลูกหนี้สิทธิ (vn_stat.uc_money > 0), มี principal diagnosis และเข้าเงื่อนไขที่เบิกชดเชยจาก สปสช. ได้อย่างน้อย 1 ข้อ ถ้าขึ้น error ชื่อคอลัมน์ ให้รัน SQL เช็คชื่อคอลัมน์ที่อยู่ท้ายไฟล์ก่อน
  4. Export ผลเป็น Excel (.xls หรือ .xlsx) ห้ามเปลี่ยนชื่อหัวคอลัมน์ ระบบหาคอลัมน์จากชื่อ vstdate และ cid คอลัมน์อื่นจะถูกคงไว้ในไฟล์ที่ดาวน์โหลดตามลำดับเดิม
  5. ดาวน์โหลด Statement OP UC (ไฟล์ STM_xxxxx_OPUCSyyyymm_nn.xls) จากระบบ สปสช. ให้ครบทุกงวดที่ครอบคลุมช่วงวันที่ลูกหนี้ จะรวมเป็น .zip ก็ได้
  6. อัปโหลดไฟล์ลูกหนี้ (ขั้นที่ 1) และไฟล์ STM (ขั้นที่ 2) แล้วกด “จับคู่” ระบบจับคู่ด้วยเลขบัตรประชาชน + วันที่รับบริการ
  7. ตรวจผล ใช้ปุ่ม ▾ ที่หัวคอลัมน์กรองตาม check_result, dup_visit, condition_found ฯลฯ แถบตัวเลขเหนือตารางสรุปจำนวน visit และจำนวนคนที่ไม่พบ STM ตามตัวกรอง แล้วกด “ดาวน์โหลด Excel” ชีต “สรุป” จะอยู่หน้าแรก
คอลัมน์ที่ได้ความหมาย
vstdateวันที่รับบริการ (ใช้จับคู่)
hn, vnHN และ VN ของ visit (VN ใช้จับคู่กับ SEQ NO ใน STM ก่อนเป็นรอบแรก)
cidเลขบัตรประชาชน (ใช้จับคู่)
auth_codeAuthen Code จาก visit_pttype ถ้ามีหลายสิทธิ คั่นด้วย ,
condition_foundเงื่อนไขที่ทำให้ visit นี้คาดว่าเบิกได้ เช่น ยา TTMT 7 หลัก, Instrument, Project code, P&P, DRUGP/TELMED, FP
check_result“ครบถ้วน” หรือ “ไม่ครบ: …” บอกรหัส ADP หรือ ICD10 ที่ยังขาด
dup_visitHN นี้มีหลาย visit ในวันเดียวกัน (นับทุก visit ใน ovst)
pttype, hipdata_codeสิทธิล่าสุดของ visit (จาก visit_pttype ที่อัปเดตล่าสุด) และรหัสกองทุน (UCS / WEL)
SQL: ลูกหนี้ OPD ที่คาดว่าเบิกชดเชยจาก สปสช. ได้ + ตรวจความครบถ้วนเวชระเบียน 
/* =====================================================================
   ลูกหนี้ OPD ที่คาดว่าเบิกชดเชยจากกองทุน สปสช. ได้ + ตรวจความครบถ้วนเวชระเบียน
   HOSxP XE (PostgreSQL)
   ใช้ชนกับ Statement สปสช. ด้วย เลขบัตรประชาชน + วันที่รับบริการ
   >>> แก้ช่วงวันที่ใน CTE params <<<
   ===================================================================== */
WITH params AS (
    SELECT DATE '2026-10-01' AS d1,
           DATE '2026-10-31' AS d2
),

/* 0) สิทธิล่าสุดที่เพิ่มใน visit (visit_pttype) — แก้ปัญหาเปลี่ยนสิทธิภายหลัง
      เรียงจาก update_datetime ล่าสุด ถ้าเท่ากัน/ว่าง ใช้ pttype_number มากสุด
      ไม่มีแถวใน visit_pttype → ใช้ ovst.pttype                                   */
vp_last AS (
    SELECT DISTINCT ON (vp.vn) vp.vn, vp.pttype
    FROM visit_pttype vp
    JOIN ovst o ON o.vn = vp.vn
    CROSS JOIN params pr
    WHERE o.vstdate BETWEEN pr.d1 AND pr.d2
      AND COALESCE(btrim(vp.pttype), '') <> ''
    ORDER BY vp.vn, vp.update_datetime DESC NULLS LAST, vp.pttype_number DESC NULLS LAST
),

/* 0.1) จำนวน visit ต่อ HN ต่อวัน (นับครั้งเดียวทั้งช่วง แทนการนับทีละแถว) */
vday AS (
    SELECT o.hn, o.vstdate, count(*) AS n
    FROM ovst o
    CROSS JOIN params pr
    WHERE o.vstdate BETWEEN pr.d1 AND pr.d2
    GROUP BY o.hn, o.vstdate
),

/* 1) visit OPD ของคนไทย (nationality = 99) ในช่วงวันที่ ไม่รวม visit ที่ admit (ovst.an มีค่า)
      และไม่รวม visit ที่ลูกหนี้สิทธิ (vn_stat.uc_money) = 0
      และต้องมี principal diagnosis (ovstdiag.diagtype = '1')
      pttype = สิทธิล่าสุดจาก vp_last */
v AS (
    SELECT o.vn, o.hn, o.vstdate,
           COALESCE(vl.pttype, o.pttype) AS pttype,
           p.cid, p.fname, p.lname,
           (p.sex = '2') AS fem,
           COALESCE(vs.age_y,
                    date_part('year', age(o.vstdate, p.birthday))::int) AS age_y,
           /* อายุเป็นเดือน (จากวันเกิด ณ วันรับบริการ) */
           (date_part('year', age(o.vstdate, p.birthday)) * 12
            + date_part('month', age(o.vstdate, p.birthday)))::int    AS age_mo,
           upper(COALESCE(pt.hipdata_code, ''))                       AS hipdata_code,
           COALESCE(vs.uc_money, 0)                                   AS uc_money,
           /* จำนวน visit ของ HN นี้ในวันเดียวกัน (ทุก visit ใน ovst) */
           COALESCE(vd.n, 1)                                          AS n_visit_day
    FROM ovst o
    CROSS JOIN params pr
    JOIN patient p       ON p.hn = o.hn
    LEFT JOIN vn_stat vs ON vs.vn = o.vn
    LEFT JOIN vp_last vl ON vl.vn = o.vn
    LEFT JOIN vday vd    ON vd.hn = o.hn AND vd.vstdate = o.vstdate
    LEFT JOIN pttype pt  ON pt.pttype = COALESCE(vl.pttype, o.pttype)
    WHERE o.vstdate BETWEEN pr.d1 AND pr.d2
      AND ltrim(p.nationality, '0') = '99'  /* คนไทย: รับทั้ง '99' และ '099' */
      AND COALESCE(btrim(o.an), '') = ''          /* ไม่นับ visit ที่ admit */
      AND COALESCE(vs.uc_money, 0) > 0            /* ไม่นับ visit ที่ลูกหนี้สิทธิ = 0 */
      AND EXISTS (SELECT 1 FROM ovstdiag pd       /* ต้องมี principal diagnosis */
                  WHERE pd.vn = o.vn AND pd.diagtype = '1'
                    AND COALESCE(btrim(pd.icd10), '') <> '')
),

/* 4.3) รายการรหัส Instrument (ADP type 2) ที่นับ */
inst AS (
    SELECT ARRAY[
           '0604','1001','1003','1007','1008','1009','1101','1102','1104','2001','2002','2004',
           '2005','2006','2007','2101','2102','2103','2104','2105','2106','2113','2115','2702',
           '2704','2704B','2704D','3001','3002','3003','3004','3005','3006','3101','4003','4004',
           '4006','4007','4008','4009','4010','4012','4013','4014','4015','4016','4024','4101',
           '4102','4103','4104','4107','4108','4109','4115','4116','4301','4302','4303','4306',
           '4307','4308','4309','4310','4312','4313','4314','4316','4319','4320','4321','4401',
           '4402','4403','4406','4407','4408','4409','4411','4415','4416','4417','4418','4419',
           '4420','4421','4423','4424','4425','4428','4501','4502','4503','4504','4505','4506',
           '4507','4508','4509','4510','4511','4601','4602','4603','4604','4605','4701','4702',
           '4704','4705','4712','4716','4721','4801','4802','4803','4804','4810','4817','4836',
           '5101','5103','5104','5105','5106','5201','5201B','5203','5204','5301','5302','5303',
           '5501','5502','5503','5608','5611','5701','5702','5703A','5703B','5704A','5704B','5704C',
           '5704D','6001','6002','6204','7002','7003','7004A','7004B','7005A','7005B','7006','7008',
           '7010','7011','7019','7027','7028','7029','7030','7102','7105','7201','7202','7203',
           '7204','7205','7206','7207A','7207B','7208A','7208B','7209','7210','7211','7212','7213',
           '7214A','7214B','7215','7216','7224','7225','7227','7301','7302','7303','7304','7305',
           '7306','7307','7308','7311','7313','7314','7315','7401','7402','7502','7506','7601',
           '8301','8305','8306','8307','8401','8402','8403','8501','8502','8503','8506','8507',
           '8510','8512','8514','8519','8520','8601','8602','8603','8604','8605','8606','8609',
           '8610','8611','8612','8701','8702','8703','8704','8705','8813','8814','8815','9105']::text[] AS codes
),

/* 5) ฐานข้อมูลต่อ visit — ค้นจากตารางจริงทีละ visit ผ่าน index (LATERAL)
      ADP / ยา TTMT / ICD10 / Authen Code */
b AS (
    SELECT v.*,
           v.hipdata_code IN ('UCS', 'WEL') AS uw,
           COALESCE(dx.dxs, '')             AS dxs,
           COALESCE(a.c_all, '{}'::text[])  AS c_all,
           COALESCE(a.c2,    '{}'::text[])  AS c2,
           COALESCE(a.c4,    '{}'::text[])  AS c4,
           COALESCE(a.c5,    '{}'::text[])  AS c5,
           COALESCE(t.n, 0)                 AS ttmt_n,
           au.auth_code
    FROM v

    /* รายการค่ารักษาที่ผูก NHSO ADP code */
    LEFT JOIN LATERAL (
        SELECT array_agg(DISTINCT x.code)                                 AS c_all,
               array_agg(DISTINCT x.code) FILTER (WHERE x.adp_type = '2') AS c2,  /* Instrument */
               array_agg(DISTINCT x.code) FILTER (WHERE x.adp_type = '4') AS c4,  /* ส่งเสริมป้องกัน */
               array_agg(DISTINCT x.code) FILTER (WHERE x.adp_type = '5') AS c5   /* Project code */
        FROM (SELECT upper(btrim(n.nhso_adp_code))   AS code,
                     btrim(n.nhso_adp_type_id::text) AS adp_type
              FROM opitemrece r
              JOIN nondrugitems n ON n.icode = r.icode
              WHERE r.vn = v.vn
                AND COALESCE(btrim(n.nhso_adp_code), '') <> ''
                AND COALESCE(r.qty, 0) > 0) x
    ) a ON true

    /* ยาที่มี TTMT code 7 หลัก */
    LEFT JOIN LATERAL (
        SELECT count(DISTINCT r.icode) AS n
        FROM opitemrece r
        JOIN drugitems d ON d.icode = r.icode
        WHERE r.vn = v.vn
          AND btrim(d.ttmt_code) ~ '^[0-9]{7}$'
          AND COALESCE(r.qty, 0) > 0
    ) t ON true

    /* ICD10 ของ visit (ตัดจุด/ตัวใหญ่) รูป ',Z130,Z321,' */
    LEFT JOIN LATERAL (
        SELECT ',' || string_agg(DISTINCT upper(replace(btrim(d.icd10), '.', '')), ',') || ',' AS dxs
        FROM ovstdiag d
        WHERE d.vn = v.vn
          AND COALESCE(btrim(d.icd10), '') <> ''
    ) dx ON true

    /* Authen Code — ถ้ามีหลายสิทธิ รวมด้วย , */
    LEFT JOIN LATERAL (
        SELECT string_agg(DISTINCT btrim(vp.auth_code), ', ') AS auth_code
        FROM visit_pttype vp
        WHERE vp.vn = v.vn
          AND COALESCE(btrim(vp.auth_code), '') <> ''
    ) au ON true
),

/* 6) เงื่อนไขที่ตรวจพบ: vn | ลำดับ | เงื่อนไข | code | สิ่งที่ขาด ('' = ครบ) */
hits AS (
    /* ก. สิทธิ 03 05 31 ทุกราย */
    SELECT vn, 1 AS ord, 'สิทธิ ' || pttype || ' (ทุกราย)' AS cond, ''::text AS code, ''::text AS miss
    FROM b WHERE pttype IN ('03', '05', '31')

    /* 1. ยา TTMT 7 หลัก */
    UNION ALL
    SELECT vn, 10, 'ยา TTMT 7 หลัก (' || ttmt_n || ' รายการ)', '', ''
    FROM b WHERE uw AND ttmt_n > 0

    /* 2. Instrument (type 2) — นับเฉพาะ nhso_adp_code ที่อยู่ในรายการที่กำหนด (240 รหัส) */
    UNION ALL
    SELECT vn, 20, 'Instrument: ' || array_to_string(x, ','), '', ''
    FROM (SELECT vn,
                 ARRAY(SELECT c FROM unnest(c2) c WHERE c = ANY(inst.codes) ORDER BY c) AS x
          FROM b CROSS JOIN inst WHERE uw) s
    WHERE cardinality(x) > 0

    /* 3. Project code (type 5) — WALKIN นับเฉพาะสิทธิ 05 และ 31 · ไม่นับ HD3995 */
    UNION ALL
    SELECT vn, 30, 'Project code: ' || array_to_string(x, ','), '', ''
    FROM (SELECT vn,
                 array_remove(CASE WHEN pttype IN ('05', '31') THEN c5 ELSE array_remove(c5, 'WALKIN') END,
                              'HD3995') AS x
          FROM b WHERE uw) s
    WHERE cardinality(x) > 0

    /* 4. P&P (type 4) — ประมวลผลเฉพาะรหัสที่กำหนด:
          13001 14001 15001 30001 30009 30010 30011 30012 30013 30014 30015 30016 90004 90005
          รหัสที่ไม่มีเงื่อนไขเฉพาะ (90004) นับตรงนี้ ไม่ต้อง check
          รหัสที่มีเงื่อนไขเฉพาะ อยู่ด้านล่าง                                           */
    UNION ALL
    SELECT vn, 40, 'P&P: ' || array_to_string(x, ','), '', ''
    FROM (SELECT vn,
                 ARRAY(SELECT c FROM unnest(c4) c
                       WHERE c IN ('90004')
                       ORDER BY c) AS x
          FROM b WHERE uw) s
    WHERE cardinality(x) > 0

    /* 13001: หญิง 13-24 ปี (หรือหญิง 13-24 ปีที่มี 30101) | ต้องมี 13001 + ICD10 */
    UNION ALL
    SELECT vn, 41, 'P&P 13001', '13001',
           concat_ws(', ', CASE WHEN NOT '13001' = ANY(c_all) THEN 'ขาด ADP 13001' END,
                           CASE WHEN dxs = '' THEN 'ไม่มี ICD10' END)
    FROM b WHERE uw AND fem AND age_y BETWEEN 13 AND 24
             AND ('13001' = ANY(c_all) OR '30101' = ANY(c_all))

    /* 14001: หญิง 13-45 ปี | ต้องมี Z130 */
    UNION ALL
    SELECT vn, 42, 'P&P 14001', '14001',
           concat_ws(', ', CASE WHEN dxs !~ ',Z130' THEN 'ไม่มี ICD10 Z130' END)
    FROM b WHERE uw AND fem AND age_y BETWEEN 13 AND 45 AND '14001' = ANY(c_all)

    /* 15001: อายุ 25-59 ปี | ต้องมี ICD10 */
    UNION ALL
    SELECT vn, 43, 'P&P 15001', '15001',
           concat_ws(', ', CASE WHEN dxs = '' THEN 'ไม่มี ICD10' END)
    FROM b WHERE uw AND age_y BETWEEN 25 AND 59 AND '15001' = ANY(c_all)

    /* 30009 / 30010: หญิง */
    UNION ALL
    SELECT vn, 44, 'P&P 30009', '30009', ''
    FROM b WHERE uw AND fem AND '30009' = ANY(c_all)
    UNION ALL
    SELECT vn, 45, 'P&P 30010', '30010', ''
    FROM b WHERE uw AND fem AND '30010' = ANY(c_all)

    /* 30011: หญิง (หรือหญิงที่มี Z34-Z35 / Oxx) | ต้องมี 30011 + (Z34-Z35 หรือ Oxx) */
    UNION ALL
    SELECT vn, 46, 'P&P 30011', '30011',
           concat_ws(', ', CASE WHEN NOT '30011' = ANY(c_all) THEN 'ขาด ADP 30011' END,
                           CASE WHEN dxs !~ ',(Z3[45]|O)' THEN 'ไม่มี ICD10 Z34-Z35/Oxx' END)
    FROM b WHERE uw AND fem AND ('30011' = ANY(c_all) OR dxs ~ ',(Z3[45]|O)')

    /* 30012: หญิง ที่มี 30213+36318+30101+36350+36003 ครบทุกตัว | ต้องมี 30012 */
    UNION ALL
    SELECT vn, 47, 'P&P 30012', '30012',
           concat_ws(', ', CASE WHEN NOT '30012' = ANY(c_all) THEN 'ขาด ADP 30012' END)
    FROM b WHERE uw AND fem
             AND c_all @> ARRAY['30213','36318','30101','36350','36003']

    /* 30013: หญิง ที่มี 36350+36003 และต้องไม่มี 36318 | ต้องมี 30013 */
    UNION ALL
    SELECT vn, 48, 'P&P 30013', '30013',
           concat_ws(', ', CASE WHEN NOT '30013' = ANY(c_all) THEN 'ขาด ADP 30013' END)
    FROM b WHERE uw AND fem
             AND c_all @> ARRAY['36350','36003']
             AND NOT '36318' = ANY(c_all)

    /* 30014: หญิง (หรือหญิงที่มี 31101) | ต้องมี 30014 + Z320/Z321 */
    UNION ALL
    SELECT vn, 49, 'P&P 30014', '30014',
           concat_ws(', ', CASE WHEN NOT '30014' = ANY(c_all) THEN 'ขาด ADP 30014' END,
                           CASE WHEN dxs !~ ',Z32[01]' THEN 'ไม่มี ICD10 Z320/Z321' END)
    FROM b WHERE uw AND fem AND ('30014' = ANY(c_all) OR '31101' = ANY(c_all))

    /* 30015: หญิง | ต้องมี Z390/Z391/Z392 */
    UNION ALL
    SELECT vn, 50, 'P&P 30015', '30015',
           concat_ws(', ', CASE WHEN dxs !~ ',Z39[012]' THEN 'ไม่มี ICD10 Z390/Z391/Z392' END)
    FROM b WHERE uw AND fem AND '30015' = ANY(c_all)

    /* 30016: หญิง | ต้องมี Z390/Z391/Z392 — ยกเว้นถ้ามี 30011 ใน visit เดียวกัน ไม่ต้อง check */
    UNION ALL
    SELECT vn, 51, 'P&P 30016', '30016',
           concat_ws(', ', CASE WHEN NOT '30011' = ANY(c_all) AND dxs !~ ',Z39[012]'
                                THEN 'ไม่มี ICD10 Z390/Z391/Z392' END)
    FROM b WHERE uw AND fem AND '30016' = ANY(c_all)

    /* 90005: อายุ 50-70 ปี (หรือที่มี 31203) | ต้องมี 90005 + Z121 */
    UNION ALL
    SELECT vn, 52, 'P&P 90005', '90005',
           concat_ws(', ', CASE WHEN NOT '90005' = ANY(c_all) THEN 'ขาด ADP 90005' END,
                           CASE WHEN dxs !~ ',Z121' THEN 'ไม่มี ICD10 Z121' END)
    FROM b WHERE uw AND age_y BETWEEN 50 AND 70
             AND ('90005' = ANY(c_all) OR '31203' = ANY(c_all))

    /* 30001: นับเฉพาะ visit ที่มี Z515 + Z718 และ diagnosis อย่างน้อย 1 ตัวใน
       C00-C96, I60-I69, F03, N185, I50, S06, T00-T07, B20-B24 | ไม่ต้อง check */
    UNION ALL
    SELECT vn, 53, 'P&P 30001', '30001', ''
    FROM b WHERE uw AND '30001' = ANY(c_all)
             AND dxs ~ ',Z515' AND dxs ~ ',Z718'
             AND dxs ~ ',(C[0-8][0-9]|C9[0-6]|I6[0-9]|F03|N185|I50|S06|T0[0-7]|B2[0-4])'

    /* คัดกรองโลหิตจางจากขาดธาตุเหล็ก (เด็ก ทุกเพศ): อายุ 6-12 เดือน หรือ 3-5 ปี (36-71 เดือน)
       13001 เด็ก: มี 30101 / 30102 / 13001 | check: (30101 หรือ 30102) + 13001 + Z130
       14001 เด็ก: มี 14001                 | check: 14001 + Z130 */
    UNION ALL
    SELECT vn, 54, 'P&P 13001 (เด็ก คัดกรองโลหิตจาง)', '13001',
           concat_ws(', ', CASE WHEN NOT c_all && ARRAY['30101','30102'] THEN 'ขาด ADP 30101/30102' END,
                           CASE WHEN NOT '13001' = ANY(c_all) THEN 'ขาด ADP 13001' END,
                           CASE WHEN dxs !~ ',Z130' THEN 'ไม่มี ICD10 Z130' END)
    FROM b WHERE uw AND (age_mo BETWEEN 6 AND 12 OR age_mo BETWEEN 36 AND 71)
             AND c_all && ARRAY['30101','30102','13001']
    UNION ALL
    SELECT vn, 55, 'P&P 14001 (เด็ก คัดกรองโลหิตจาง)', '14001',
           concat_ws(', ', CASE WHEN dxs !~ ',Z130' THEN 'ไม่มี ICD10 Z130' END)
    FROM b WHERE uw AND (age_mo BETWEEN 6 AND 12 OR age_mo BETWEEN 36 AND 71)
             AND '14001' = ANY(c_all)

    /* HCV viral load: มี 36333 | check: มี ICD10 B182
       APRI score: มี ICD10 B182 + 30101 + (32003 หรือ 32310) | check: มี ADP 3633B */
    UNION ALL
    SELECT vn, 56, 'HCV viral load 36333', '36333',
           CASE WHEN dxs !~ ',B182' THEN 'ไม่มี ICD10 B182' ELSE '' END
    FROM b WHERE uw AND '36333' = ANY(c_all)
    UNION ALL
    SELECT vn, 57, 'APRI score 3633B', '3633B',
           CASE WHEN NOT '3633B' = ANY(c_all) THEN 'ขาด ADP 3633B' ELSE '' END
    FROM b WHERE uw AND dxs ~ ',B182' AND '30101' = ANY(c_all)
             AND c_all && ARRAY['32003','32310']

    /* 5. type 3 — 1B004N / 1B004P | ต้องมี Z124 หรือ Z014 */
    UNION ALL
    SELECT vn, 60, 'Other: ' || array_to_string(x, ','), array_to_string(x, '/'),
           CASE WHEN dxs !~ ',Z(124|014)' THEN 'ไม่มี ICD10 Z124/Z014' ELSE '' END
    FROM (SELECT vn, dxs,
                 ARRAY(SELECT c FROM unnest(c_all) c WHERE c IN ('1B004N','1B004P') ORDER BY c) AS x
          FROM b WHERE uw) s
    WHERE cardinality(x) > 0

    /* Cons01 / Eva001: นับเฉพาะ visit ที่มีครบทั้ง Cons01 + Eva001 + Z515 + Z718
       และมี diagnosis อย่างน้อย 1 ตัวใน C00-C96, I60-I69, F03, N185, I50, S06, T00-T07, B20-B24
       ไม่ต้อง check */
    UNION ALL
    SELECT vn, 61, 'Other: Cons01/Eva001', 'Cons01/Eva001', ''
    FROM b WHERE uw
             AND c_all @> ARRAY['CONS01','EVA001']
             AND dxs ~ ',Z515' AND dxs ~ ',Z718'
             AND dxs ~ ',(C[0-8][0-9]|C9[0-6]|I6[0-9]|F03|N185|I50|S06|T0[0-7]|B2[0-4])'

    /* DRUGP / TELMED */
    UNION ALL
    SELECT vn, 62, 'Other: ' || array_to_string(x, ','), '', ''
    FROM (SELECT vn,
                 ARRAY(SELECT c FROM unnest(c_all) c WHERE c IN ('DRUGP','TELMED') ORDER BY c) AS x
          FROM b WHERE uw) s
    WHERE cardinality(x) > 0

    /* FP001 | ต้องมี Z301 */
    UNION ALL
    SELECT vn, 63, 'Other: FP001', 'FP001',
           CASE WHEN dxs !~ ',Z301' THEN 'ไม่มี ICD10 Z301' ELSE '' END
    FROM b WHERE uw AND 'FP001' = ANY(c_all)

    /* FP002_1 / FP002_2 | ต้องมี Z308 */
    UNION ALL
    SELECT vn, 64, 'Other: ' || array_to_string(x, ','), array_to_string(x, '/'),
           CASE WHEN dxs !~ ',Z308' THEN 'ไม่มี ICD10 Z308' ELSE '' END
    FROM (SELECT vn, dxs,
                 ARRAY(SELECT c FROM unnest(c_all) c WHERE c IN ('FP002_1','FP002_2') ORDER BY c) AS x
          FROM b WHERE uw) s
    WHERE cardinality(x) > 0
)

/* 7) ผลลัพธ์: 1 แถวต่อ visit */
SELECT b.vstdate                                     AS vstdate,
       b.hn                                          AS hn,
       b.cid                                         AS cid,
       b.auth_code::varchar(200)                     AS auth_code,
       b.fname                                       AS fname,
       b.lname                                       AS lname,
       string_agg(h.cond, ' | ' ORDER BY h.ord)::varchar(1000) AS condition_found,
       (CASE WHEN bool_and(h.miss = '') THEN 'ครบถ้วน'
            ELSE 'ไม่ครบ: ' || string_agg(h.code || ' ' || h.miss, '; ' ORDER BY h.ord)
                               FILTER (WHERE h.miss <> '')
       END)::varchar(1000)                           AS check_result,
       (CASE WHEN b.n_visit_day > 1
            THEN 'มี visit ซ้ำในวัน (' || b.n_visit_day || ' visit)' ELSE '' END)::varchar(50) AS dup_visit,
       b.vn, b.pttype, b.hipdata_code
FROM hits h
JOIN b ON b.vn = h.vn
GROUP BY b.vstdate, b.cid, b.fname, b.lname, b.hn, b.vn, b.pttype, b.hipdata_code, b.auth_code, b.n_visit_day
ORDER BY b.vstdate, b.cid;

รายเดือน

รายชื่อลูกหนี้ที่ยังไม่พบใน STM

ผลการจับคู่จะแสดงที่นี่ ได้แก่ สรุปยอด, ประเภทของรายการที่ไม่พบ, ตารางรายเดือน และรายชื่อที่ดาวน์โหลดเป็น Excel ได้

วิธีจับคู่

  • รอบแรกจับคู่ VN ของลูกหนี้กับ SEQ NO ใน STM (ต้องมีเลขบัตรและวันที่ตรงกันด้วย) จากนั้นจับคู่เลขบัตรประชาชน + วันที่รับบริการแบบนับจำนวนครั้ง
  • ไม่ใช้ชื่อในการจับคู่ เพราะชื่อใน STM อาจสะกดต่างหรือมีข้อความต่อท้าย
  • ไม่พบใน STM: ไม่มีเลขบัตร + วันที่นี้ในทุกไฟล์ STM · หลังรอบ STM ล่าสุด: วันรับบริการหลังวันล่าสุดที่มีใน STM · มาซ้ำวันเดียวกัน: วันเดียวกันมีหลาย visit แต่ STM มีน้อยกว่า
  • พบใน STM แต่ชดเชย 0 บาท: จับคู่ได้ แต่ “ยอดชดเชยทั้งสิ้น” ของรายการ STM นั้น (รวมข้อมูลปกติ + อุทธรณ์) เป็น 0 และไม่มีรายการ STM อื่นของเลขบัตร + วันเดียวกันที่ได้ชดเชย
  • อัปโหลดไฟล์ STM ซ้ำได้ ระบบรวมรายการเดียวกันให้อัตโนมัติ
  • ไฟล์ที่ดาวน์โหลดคงคอลัมน์เดิมของไฟล์ลูกหนี้ไว้ทั้งหมดตามลำดับเดิม (เช่น condition_found, check_result) แล้วต่อท้ายด้วยผลการจับคู่
  • “Excel ฉบับเต็ม” มี 3 ชีต: ลูกหนี้ทุกรายพร้อมผลจับคู่, รายการที่ไม่พบ และรายการ STM ที่ไม่มีลูกหนี้คู่
